# The renderer, the GUI library and the scene the operator windows derive from are published surface
# here, as is the model library, so none of them is on the list. The solver and the control library
# are what an extension must never publish: the solver is linked here for the reference
# implementation and confined to it, and the logging dependency is confined the same way.
praxis_assert_no_include_leak(praxis_manipulator cartan ctrlpp spdlog)

set(PRAXIS_PUBLIC_DEPENDENCIES meios threepp imgui)

# Every dependency this extension has, it declares, and nothing first-party sits above it: there is
# no absence left to claim here that could ever fail. NONE says so explicitly, because an empty list
# reads the same as one a later author forgot to fill in.
set(PRAXIS_CLOSURE_ABSENT NONE)

# NAMESPACE prefixes the target name rather than the alias, so without EXPORT_NAME a consumer
# writing praxis::manipulator is told about a target it never named.
set_target_properties(praxis_manipulator PROPERTIES EXPORT_NAME manipulator)
